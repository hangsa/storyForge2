```python
"""AutopilotSession — v1.9 server-driven orchestration model.

Spec: docs/design/storyForge-design-v1.9.md §四 F1.9.1
"""
from __future__ import annotations

from dataclasses import dataclass, field, asdict
from datetime import datetime, timezone
from enum import Enum
from typing import Literal, Optional
import uuid


class SessionState(str, Enum):
    IDLE = "idle"
    RUNNING = "running"
    PAUSED = "paused"
    STOPPED = "stopped"
    ERROR = "error"


class InvalidTransition(Exception):
    pass


VALID_TRANSITIONS: dict[SessionState, dict[str, SessionState]] = {
    SessionState.IDLE:    {"start": SessionState.RUNNING, "stop": SessionState.STOPPED},
    SessionState.RUNNING: {
        "pause":          SessionState.PAUSED,
        "stop":           SessionState.STOPPED,
        "circuit_open":   SessionState.PAUSED,
        "fatal_error":    SessionState.ERROR,
        "task_complete":  SessionState.RUNNING,    # recurring event, state stays running
    },
    SessionState.PAUSED:  {"resume": SessionState.RUNNING, "stop": SessionState.STOPPED, "start": SessionState.RUNNING},
    SessionState.STOPPED: {"start": SessionState.RUNNING},
    SessionState.ERROR:   {"start": SessionState.RUNNING},  # spec: error → idle → running
}


@dataclass
class ManagedStartConfig:
    scope: Literal["all_planned", "next_chapter"] = "all_planned"
    cadence: Literal["fast", "balanced", "careful"] = "balanced"
    policy: Literal["auto", "ask"] = "auto"
    notify: Literal["all", "milestones"] = "milestones"


@dataclass
class CircuitSnapshot:
    force_pass_count: int = 0
    retry_count: int = 0
    last_event_at: Optional[str] = None


@dataclass
class CurrentTask:
    kind: Literal["plan_chapter", "write_scene", "fact_guard", "review", "archival", "diagnosis"]
    chapter_number: Optional[int]
    scene_id: Optional[str]
    status: Literal["queued", "active", "blocked", "completed", "failed"]
    started_at: Optional[str]
    description: str
    progress_pct: Optional[int] = None  # v1.9.x for scene-level granularity


@dataclass
class QueueItem:
    id: str
    kind: str
    chapter_number: Optional[int]
    scheduled_at: Optional[str]
    priority: int
    payload: dict = field(default_factory=dict)


@dataclass
class SessionEvent:
    id: str
    at: str
    type: Literal[
        "task_start", "task_complete", "task_fail", "decision", "intervention",
        "checkpoint", "circuit_open", "circuit_close", "queue_add", "queue_drop",
    ]
    task_id: Optional[str] = None
    chapter_number: Optional[int] = None
    payload: dict = field(default_factory=dict)


@dataclass
class AutopilotSession:
    project_id: str
    state: SessionState
    config: ManagedStartConfig
    started_at: Optional[str]
    last_heartbeat_at: Optional[str]
    current_task: Optional[CurrentTask]
    queue: list[QueueItem]
    history: list[SessionEvent]
    circuit: CircuitSnapshot


# --- Pure functions (return new session; do not mutate) ---

def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _new_event(ev_type: str, **kw) -> SessionEvent:
    return SessionEvent(
        id=f"evt_{uuid.uuid4().hex[:10]}",
        at=_now(),
        type=ev_type,
        **kw,
    )


def transition(session: AutopilotSession, action: str) -> AutopilotSession:
    """Pure transition. Raises InvalidTransition if (state, action) not allowed."""
    valid = VALID_TRANSITIONS.get(session.state, {})
    if action not in valid:
        raise InvalidTransition(f"cannot {action} from {session.state.value}")

    target = valid[action]
    new_history = list(session.history)

    if action == "start":
        if session.state == SessionState.ERROR:
            # Spec: error → idle → running; emit two events
            new_history.append(_new_event("task_start", payload={"from": "error"}))
        else:
            new_history.append(_new_event("task_start"))
        new_started_at = session.started_at or _now()
    else:
        new_started_at = session.started_at

    if action == "stop":
        new_history.append(_new_event("task_complete", payload={"reason": "stop_request"}))
    elif action == "circuit_open":
        new_history.append(_new_event("circuit_open"))
    elif action in ("pause", "resume"):
        new_history.append(_new_event("checkpoint", payload={"action": action}))

    return AutopilotSession(
        project_id=session.project_id,
        state=target,
        config=session.config,
        started_at=new_started_at,
        last_heartbeat_at=session.last_heartbeat_at,
        current_task=session.current_task if target != SessionState.STOPPED else None,
        queue=session.queue,
        history=new_history + ([_new_event("task_complete" if action == "task_complete" else "decision")] if action in ("task_complete",) else []),
        circuit=session.circuit,
    )


def set_current_task(session: AutopilotSession, task: CurrentTask) -> AutopilotSession:
    return AutopilotSession(
        project_id=session.project_id,
        state=session.state,
        config=session.config,
        started_at=session.started_at,
        last_heartbeat_at=session.last_heartbeat_at,
        current_task=task,
        queue=session.queue,
        history=session.history,
        circuit=session.circuit,
    )


def complete_current_task(session: AutopilotSession) -> AutopilotSession:
    return AutopilotSession(
        project_id=session.project_id,
        state=session.state,
        config=session.config,
        started_at=session.started_at,
        last_heartbeat_at=session.last_heartbeat_at,
        current_task=None,
        queue=session.queue,
        history=session.history + [_new_event("task_complete", task_id=session.current_task.scene_id if session.current_task else None)],
        circuit=session.circuit,
    )


def add_queue_item(session: AutopilotSession, item: QueueItem) -> AutopilotSession:
    return AutopilotSession(
        project_id=session.project_id,
        state=session.state,
        config=session.config,
        started_at=session.started_at,
        last_heartbeat_at=session.last_heartbeat_at,
        current_task=session.current_task,
        queue=session.queue + [item],
        history=session.history + [_new_event("queue_add", payload={"item_id": item.id})],
        circuit=session.circuit,
    )


def drop_queue_item(session: AutopilotSession, item_id: str) -> AutopilotSession:
    return AutopilotSession(
        project_id=session.project_id,
        state=session.state,
        config=session.config,
        started_at=session.started_at,
        last_heartbeat_at=session.last_heartbeat_at,
        current_task=session.current_task,
        queue=[q for q in session.queue if q.id != item_id],
        history=session.history + [_new_event("queue_drop", payload={"item_id": item_id})],
        circuit=session.circuit,
    )
```

```python
"""AutopilotSessionManager — write-through persistence for AutopilotSession.

Spec: docs/design/storyForge-design-v1.9.md §四 F1.9.1
"""
from __future__ import annotations

import json
import logging
from pathlib import Path
from typing import Optional

from backend.models.autopilot_session import (
    AutopilotSession, SessionState, ManagedStartConfig, CurrentTask, QueueItem,
    CircuitSnapshot,
    transition, set_current_task, complete_current_task,
    add_queue_item, drop_queue_item,
)

logger = logging.getLogger(__name__)


class AutopilotSessionManager:
    """Owns the on-disk file for a single project's AutopilotSession."""

    def __init__(self, projects_dir: Path, project_id: str) -> None:
        self._dir = Path(projects_dir) / project_id / "autopilot"
        self._project_id = project_id
        self._file = self._dir / "session.json"

    @property
    def session_path(self) -> Path:
        return self._file

    def _ensure_dir(self) -> None:
        self._dir.mkdir(parents=True, exist_ok=True)

    def _empty(self) -> AutopilotSession:
        return AutopilotSession(
            project_id=self._project_id,
            state=SessionState.IDLE,
            config=ManagedStartConfig(),
            started_at=None,
            last_heartbeat_at=None,
            current_task=None,
            queue=[],
            history=[],
            circuit=CircuitSnapshot(),
        )

    def load(self) -> Optional[AutopilotSession]:
        if not self._file.exists():
            return None
        try:
            raw = json.loads(self._file.read_text(encoding="utf-8"))
            return _dict_to_session(raw)
        except Exception as e:
            logger.warning("session.json unreadable for %s: %s", self._project_id, e)
            return None

    def save(self, session: AutopilotSession) -> None:
        self._ensure_dir()
        tmp = self._file.with_suffix(".json.tmp")
        tmp.write_text(
            json.dumps(_session_to_dict(session), ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        tmp.replace(self._file)

    # --- High-level intent methods (each writes through) ---

    def ensure_idle_session(self) -> AutopilotSession:
        s = self.load() or self._empty()
        if s.state == SessionState.IDLE:
            self.save(s)
        return s

    def start(self, cfg: ManagedStartConfig) -> AutopilotSession:
        cur = self.load() or self._empty()
        # Spec: "start" idempotent if session is already running — return as-is
        if cur.state == SessionState.RUNNING:
            return cur
        # Replace config if caller provides one
        cur.config = cfg
        new = transition(cur, "start")
        self.save(new)
        return new

    def stop(self) -> AutopilotSession:
        cur = self.load() or self._empty()
        if cur.state == SessionState.STOPPED:
            return cur
        new = transition(cur, "stop")
        self.save(new)
        return new

    def pause(self) -> AutopilotSession:
        cur = self.load() or self._empty()
        if cur.state in (SessionState.PAUSED, SessionState.IDLE, SessionState.STOPPED):
            return cur
        new = transition(cur, "pause")
        self.save(new)
        return new

    def resume(self) -> AutopilotSession:
        cur = self.load() or self._empty()
        if cur.state != SessionState.PAUSED:
            return cur
        new = transition(cur, "resume")
        self.save(new)
        return new

    def intervene(self, action: str) -> AutopilotSession:
        """Session-level intervention: action ∈ {"pause_immediate", "stop_current_task", "rollback_checkpoint"}."""
        cur = self.load() or self._empty()
        if action == "pause_immediate":
            return self.pause()
        if action == "stop_current_task":
            new = complete_current_task(cur)
            self.save(new)
            return new
        if action == "rollback_checkpoint":
            # Phase 2: integrate with CheckpointManager
            return self.pause()
        raise ValueError(f"unknown intervention: {action}")

    def set_current_task(self, task: CurrentTask) -> AutopilotSession:
        cur = self.load() or self._empty()
        new = set_current_task(cur, task)
        self.save(new)
        return new

    def add_queue(self, item: QueueItem) -> AutopilotSession:
        cur = self.load() or self._empty()
        new = add_queue_item(cur, item)
        self.save(new)
        return new

    def drop_queue(self, item_id: str) -> AutopilotSession:
        cur = self.load() or self._empty()
        new = drop_queue_item(cur, item_id)
        self.save(new)
        return new

    def heartbeat(self) -> AutopilotSession:
        from datetime import datetime, timezone
        cur = self.load() or self._empty()
        new = AutopilotSession(
            project_id=cur.project_id,
            state=cur.state,
            config=cur.config,
            started_at=cur.started_at,
            last_heartbeat_at=datetime.now(timezone.utc).isoformat(),
            current_task=cur.current_task,
            queue=cur.queue,
            history=cur.history,
            circuit=cur.circuit,
        )
        self.save(new)
        return new


# --- Serialization helpers (test/dict roundtrip) ---

def _session_to_dict(s: AutopilotSession) -> dict:
    return {
        "project_id": s.project_id,
        "state": s.state.value,
        "config": asdict(s.config),
        "started_at": s.started_at,
        "last_heartbeat_at": s.last_heartbeat_at,
        "current_task": asdict(s.current_task) if s.current_task else None,
        "queue": [asdict(q) for q in s.queue],
        "history": [asdict(e) for e in s.history],
        "circuit": asdict(s.circuit),
    }

    )
```

```python
class AutopilotRunner:
    def __init__(
        self,
        projects_dir: Path,
        project_id: str,
        scene_writer: Callable[[str, int, int], dict],
        mgr: Optional[AutopilotSessionManager] = None,
    ) -> None:
        self._projects_dir = Path(projects_dir)
        self._project_id = project_id
        self._scene_writer = scene_writer
        self._mgr = mgr or AutopilotSessionManager(self._projects_dir, project_id)
        self._fm = FileManager(self._projects_dir)

    @property
    def mgr(self) -> AutopilotSessionManager:
        return self._mgr

    def _require_running(self) -> None:
        s = self._mgr.load()
        if s is None or s.state != SessionState.RUNNING:
            raise RuntimeError(f"session not running (state={s.state if s else 'None'})")

    def run_next_chapter(self) -> dict:
        """Run the next planned chapter. Caller must keep session running."""
        self._require_running()
        outline = self._fm.read_json(self._project_id, "outline.json") or {}
        chapters = outline.get("chapters") or []
        if not chapters:
            raise RuntimeError("no outline defined")

        # pick first chapter not yet completed
        progress = self._fm.read_json(self._project_id, "progress.json") or {}
        done = {ch["chapter_number"] for ch in progress.get("chapters", []) if ch.get("status") == "completed"}
        next_ch = next((c for c in chapters if c["chapter_number"] not in done), None)
        if next_ch is None:
            return {"status": "all_done", "chapter": None}

        chapter_number = next_ch["chapter_number"]
        scenes = next_ch.get("scene_plan") or []
        scene_results = []
        for sp in scenes:
            scene_id = str(sp["scene_number"])
            task = CurrentTask(
                kind="write_scene",
                chapter_number=chapter_number,
                scene_id=scene_id,
                status="active",
                started_at=None,
                description=f"写作第 {chapter_number} 章第 {sp['scene_number']} 幕",
                progress_pct=None,
            )
            self._mgr.set_current_task(task)
            try:
                result = self._scene_writer(self._project_id, chapter_number, sp["scene_number"])
                scene_results.append({"scene_number": sp["scene_number"], "result": result})
            except Exception as e:
                logger.exception("scene writer raised: %s", e)
                scene_results.append({"scene_number": sp["scene_number"], "error": str(e)})
                # Bump state to paused on failure per spec circuit_open handler
                self._mgr.pause()
                break

        self._mgr.set_current_task(CurrentTask(
            kind="review",
            chapter_number=chapter_number,
            scene_id=None,
            status="active",
            started_at=None,
            description=f"复核第 {chapter_number} 章",
            progress_pct=None,
        ))
        # Mark chapter complete in progress.json (write-through so UI sees real progress)
        progress = self._fm.read_json(self._project_id, "progress.json") or {}
        for ch in progress.setdefault("chapters", []):
            if ch.get("chapter_number") == chapter_number:
                ch["status"] = "completed"
        self._fm.write_json(self._project_id, "progress.json", progress)

        self._mgr.set_current_task(None)
        return {"status": "completed", "chapter": chapter_number, "scenes": scene_results}
```

```python
"""SSE broadcaster with multi-subscriber fan-out + Last-Event-ID replay.

Spec: docs/design/storyForge-design-v1.9.md §四 F1.9.2

Each project's session has one broadcaster. Subscribers register an `asyncio.Queue`
and an optional `last_event_id`; on subscribe, any historical events with `id >
last_event_id` are pushed into their queue first, then live events follow.
"""
from __future__ import annotations

import asyncio
import logging
from collections import OrderedDict
from typing import Optional

logger = logging.getLogger(__name__)


class SSEBroadcaster:
    def __init__(self, history_limit: int = 200) -> None:
        self._history: "OrderedDict[str, dict]" = OrderedDict()
        self._history_limit = history_limit
        self._subscribers: list[asyncio.Queue] = []

    async def publish(self, event: dict) -> None:
        eid = event.get("id") or f"gen_{len(self._history)}"
        event["id"] = eid
        self._history[eid] = event
        if len(self._history) > self._history_limit:
            self._history.popitem(last=False)
        for q in list(self._subscribers):
            try:
                q.put_nowait(event)
            except asyncio.QueueFull:
                # Backpressure: drop oldest in the queue, then try again
                try:
                    q.get_nowait()
                    q.put_nowait(event)
                except Exception:
                    logger.warning("dropping subscriber (repeated full queue)")
                    self._subscribers.remove(q)

    def subscribe(self, queue: asyncio.Queue, last_event_id: Optional[str]) -> None:
        # Replay: events with id > last_event_id (lexicographically).
        replay = [e for eid, e in self._history.items() if last_event_id is None or eid > last_event_id]
        for ev in replay:
            try:
                queue.put_nowait(ev)
            except asyncio.QueueFull:
                break
        self._subscribers.append(queue)

    def unsubscribe(self, queue: asyncio.Queue) -> None:
        if queue in self._subscribers:
            self._subscribers.remove(queue)


# Per-project registry — modules import this to broadcast session events.
_REGISTRY: dict[str, SSEBroadcaster] = {}


def get_broadcaster(project_id: str) -> SSEBroadcaster:
    """Lazy-create a broadcaster for a project. One per project, lifetime = process."""
    if project_id not in _REGISTRY:
        _REGISTRY[project_id] = SSEBroadcaster()
    return _REGISTRY[project_id]
```

```typescript
export interface SessionEvent {
  id: string;
  event: string;
  payload: Record<string, unknown>;
}

const STORAGE_KEY = (pid: string) => `storyforge.autopilot.last_event_id.${pid}`;

export function useAutopilotSession(projectId: string) {
  const [events, setEvents] = useState<SessionEvent[]>([]);
  const [connected, setConnected] = useState(false);
  const esRef = useRef<EventSource | null>(null);

  useEffect(() => {
    if (!projectId) return;
    const lastId = sessionStorage.getItem(STORAGE_KEY(projectId));
    const url = `/api/v1/projects/${encodeURIComponent(projectId)}/autopilot/session/events${
      lastId ? `?last_event_id=${encodeURIComponent(lastId)}` : ""
    }`;
    // Browser-native EventSource cannot set the Last-Event-ID header. We
    // append it as a query param and let the server translate (see Task 2.2
    // Step 3 — the /events handler reads ?last_event_id= as a query fallback).
    const es = new EventSource(url);
    esRef.current = es;

    es.onopen = () => setConnected(true);
    es.onerror = () => setConnected(false);
    es.onmessage = (msg: MessageEvent) => {
      try {
        const parsed = JSON.parse(msg.data);
        const ev: SessionEvent = {
          id: (msg as any).lastEventId || parsed.id,
          event: parsed.event,
          payload: parsed.data ? JSON.parse(parsed.data) : {},
        };
        sessionStorage.setItem(STORAGE_KEY(projectId), ev.id);
        setEvents((prev) => [...prev.slice(-199), ev]);  // keep last 200
      } catch {
        // ignore malformed events
      }
    };

    return () => {
      es.close();
      esRef.current = null;
    };
  }, [projectId]);

  return { events, connected };
}
```

```typescript
const API_BASE = "/api/v1/projects";

export interface ManagedStartConfig {
  scope: "all_planned" | "next_chapter";
  cadence: "fast" | "balanced" | "careful";
  policy: "auto" | "ask";
  notify: "all" | "milestones";
}

export interface AutopilotSession {
  project_id: string;
  state: "idle" | "running" | "paused" | "stopped" | "error";
  config: ManagedStartConfig;
  started_at: string | null;
  last_heartbeat_at: string | null;
  current_task: { description: string } | null;
  queue: Array<{ id: string; kind: string; chapter_number: number | null }>;
  history: Array<{ id: string; type: string; at: string }>;
}

export const AutopilotApi = {
  getSession: async (projectId: string): Promise<AutopilotSession> => {
    const r = await fetch(`${API_BASE}/${encodeURIComponent(projectId)}/autopilot/session`, {
      credentials: "same-origin",
    });
    const body = await r.json();
    if (body.detail) return body.detail;
    throw new Error(body.message ?? "session not found");
  },

  startSession: async (projectId: string, config: ManagedStartConfig): Promise<AutopilotSession> => {
    const r = await fetch(`${API_BASE}/${encodeURIComponent(projectId)}/autopilot/session/start`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(config),
    });
    const body = await r.json();
    if (body.error) throw new Error(body.message ?? "start failed");
    return body.detail;
  },

  stopSession: async (projectId: string): Promise<AutopilotSession> => {
    const r = await fetch(`${API_BASE}/${encodeURIComponent(projectId)}/autopilot/session/stop`, { method: "POST" });
    const body = await r.json();
    if (body.error) throw new Error(body.message ?? "stop failed");
    return body.detail;
  },

  pauseSession: (projectId: string) =>
    fetch(`${API_BASE}/${encodeURIComponent(projectId)}/autopilot/session/pause`, { method: "POST" }).then(r => r.json()),

  resumeSession: (projectId: string) =>
    fetch(`${API_BASE}/${encodeURIComponent(projectId)}/autopilot/session/resume`, { method: "POST" }).then(r => r.json()),

  intervene: (projectId: string, action: "pause_immediate" | "stop_current_task" | "rollback_checkpoint") =>
    fetch(`${API_BASE}/${encodeURIComponent(projectId)}/autopilot/session/intervene`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ action }),
    }).then(r => r.json()),
};

export default AutopilotApi;
```

```typescript
import { useEffect, useState } from "react";
import AutopilotApi, { ManagedStartConfig, AutopilotSession } from "../api/autopilot";

const DEFAULT_CFG: ManagedStartConfig = {
  scope: "all_planned",
  cadence: "balanced",
  policy: "auto",
  notify: "milestones",
};

export function useAutopilotConfig(projectId: string) {
  const [session, setSession] = useState<AutopilotSession | null>(null);
  const [config, setConfig] = useState<ManagedStartConfig | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (!projectId) return;
    let cancelled = false;
    AutopApi_get(projectId)
      .then((s) => {
        if (cancelled) return;
        setSession(s);
        setConfig(s.config ?? DEFAULT_CFG);
      })
      .catch(() => {
        if (cancelled) return;
        setConfig(DEFAULT_CFG);
      })
      .finally(() => !cancelled && setLoading(false));
    return () => { cancelled = true; };
  }, [projectId]);

  async function startSession(cfg: ManagedStartConfig) {
    const s = await AutopilotApi.startSession(projectId, cfg);
    setSession(s);
    setConfig(s.config);
    return s;
  }

  return { session, config, loading, startSession };
}

async function AutopApi_get(projectId: string): Promise<AutopilotSession> {
  return AutopilotApi.getSession(projectId);
}
```

```tsx
export default function ManagedAIControlPanel({ sessionState, events, queue = [], onIntervene }: Props) {
  const [tab, setTab] = useState<Tab>("decisions");
  return (
    <div data-testid="ai-control-panel" className="h-full flex flex-col">
      <div className="flex border-b border-outline-variant">
        {TABS.map((t) => (
          <button
            key={t.id}
            type="button"
            data-testid={`ai-tab-${t.id}`}
            onClick={() => setTab(t.id)}
            className={`flex-1 py-2 text-sm font-body-ui border-b-2 transition-colors ${
              tab === t.id ? "border-primary-container text-primary-container" : "border-transparent text-system-log hover:text-primary"
            }`}
          >{t.label}</button>
        ))}
      </div>
      <div className="flex-1 p-4 overflow-y-auto text-sm font-body-ui text-system-log space-y-2">
        {tab === "decisions" && <DecisionsList events={events} />}
        {tab === "queue" && <QueueList queue={queue} />}
        {tab === "checks" && <ChecksList events={events} />}
        {tab === "intervene" && <IntervenePanel sessionState={sessionState} onIntervene={onIntervene} />}
      </div>
    </div>
  );
}

function DecisionsList({ events }: { events: SessionEvent[] }) {
  return (
    <ul data-testid="ai-decisions-list" className="space-y-2">
      {events.length === 0 && <li className="p-2 rounded bg-surface-container-low/40 italic">— 暂无事件 —</li>}
      {events.slice(-20).reverse().map((e) => (
        <li key={e.id} className="p-2 rounded bg-surface-container-low">
          {e.event} {JSON.stringify(e.payload)}
        </li>
      ))}
    </ul>
  );
}

function QueueList({ queue }: { queue: QueueItem[] }) {
  return (
    <ul data-testid="ai-queue-list" className="space-y-2">
      {queue.length === 0 && <li className="p-2 rounded bg-surface-container-low/40 italic">— 队列空 —</li>}
      {queue.map((q) => (
        <li key={q.id} className="p-2 rounded bg-surface-container-low">
          计划任务 · {q.kind} 第 {q.chapter_number ?? "?"} 章
        </li>
      ))}
    </ul>
  );
}

function ChecksList({ events }: { events: SessionEvent[] }) {
  const fails = events.filter((e) => e.event === "task_fail");
  return (
    <ul data-testid="ai-checks-list" className="space-y-2">
      {fails.length === 0 && <li className="p-2 rounded bg-surface-container-low">— 最近无失败 —</li>}
      {fails.slice(-10).reverse().map((e) => (
        <li key={e.id} className="p-2 rounded bg-surface-container-low text-amber-700">
          检查失败: {JSON.stringify(e.payload)}
        </li>
      ))}
    </ul>
  );
}

function IntervenePanel({ sessionState, onIntervene }: { sessionState: SessionState; onIntervene: Props["onIntervene"] }) {
  const enabled = sessionState === "running";
  return (
    <div data-testid="ai-intervene-actions" className="space-y-2">
      <ActionButton testid="action-pause" label="暂停托管" disabled={!enabled} onClick={() => onIntervene("pause_immediate")} />
      <ActionButton testid="action-rollback" label="回滚到上一节点" disabled={!enabled} onClick={() => onIntervene("rollback_checkpoint")} />
      <ActionButton testid="action-stop" label="停止当前任务" disabled={!enabled} onClick={() => onIntervene("stop_current_task")} />
    </div>
  );
}

function ActionButton({ testid, label, disabled, onClick }: { testid: string; label: string; disabled: boolean; onClick: () => void }) {
  return (
    <button
      type="button"
      data-testid={testid}
      disabled={disabled}
      onClick={onClick}
      className={`w-full px-3 py-2 rounded-lg text-sm ${
        disabled ? "bg-surface-container text-system-log/50 cursor-not-allowed" : "bg-tertiary-container text-surface-container-low hover:opacity-90"
      }`}
    >
      {label}
    </button>
  );
}
```

```tsx
import { useEffect, useState } from "react";

  projectId: string;
  projectName: string;
  mode: WorkspaceMode;

  sessionState: "idle" | "running" | "paused" | "stopped" | "error" | null;

interface ProgressCount { completed: number; total: number; }

export default function WorkspaceTopBar({ projectId, projectName, mode, onModeChange, sessionState }: Props) {
  const [counts, setCounts] = useState<ProgressCount>({ completed: 0, total: 0 });

  useEffect(() => {
    if (mode !== "managed" || !projectId) return;
    let cancelled = false;
    const refresh = () =>
      api.getStage4Progress(projectId).then((p) => {
        if (cancelled) return;
        const chapters = (p.chapters ?? []) as Array<{ status: string }>;
        const completed = chapters.filter((c) => c.status === "completed").length;
        setCounts({ completed, total: p.total_chapters ?? 0 });
      });
    refresh();
    const interval = setInterval(refresh, 5000);
    return () => { cancelled = true; clearInterval(interval); };
  }, [projectId, mode]);

  return (
```
