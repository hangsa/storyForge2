## Task 1: `SSEBroadcaster` — backend fan-out + Last-Event-ID replay

**Files:**
- Create: `backend/utils/sse_broadcaster.py`
- Create: `tests/test_sse_broadcaster.py`

**Context:** The Stage 1 `AutopilotSession` (built in a sibling plan) exposes events as they happen — tasks start/complete, circuit opens/closes, queue mutations. Stage 2 needs a transport layer between the session publisher and N frontend subscribers (each in their own browser tab). The broadcaster must:

1. Accept `publish(event)` from any code path (Conductor, CircuitBreaker, Stage 4 queue).
2. Maintain a per-event monotonically-increasing ID.
3. Hold the last N events (default 256) so a late-joining subscriber can replay.
4. Track each subscriber as an `asyncio.Queue(maxsize=64)`. A subscriber that lags more than 64 events is dropped (slow-subscriber protection).
5. Emit a heartbeat (`:hb`) SSE comment every 30s on every open channel so proxies don't time the connection out.

Data model:

```python
@dataclass
class SSEEvent:
    id: int
    event: str            # "decision" / "task_start" / "task_complete" / etc.
    data: dict            # JSON-serializable payload
    created_at: float     # monotonic timestamp

class SSEBroadcaster:
    def __init__(self, history_size: int = 256, queue_max: int = 64): ...
    def publish(self, event: str, data: dict) -> int: ...    # returns assigned id
    async def subscribe(self, last_event_id: int | None) -> AsyncIterator[SSEEvent]: ...
    @property
    def history(self) -> list[SSEEvent]: ...
```

```python
"""Multi-subscriber Server-Sent-Events broadcaster for AutopilotSession.

Each subscriber owns an asyncio.Queue. A background task pushes events into
each subscriber's queue. Heartbeats are sent every HEARTBEAT_INTERVAL
seconds as SSE comments so reverse proxies do not idle out the connection.
"""
from __future__ import annotations

import asyncio
import time
from dataclasses import dataclass, field
from typing import AsyncIterator, Optional

HEARTBEAT_INTERVAL = 30.0  # seconds


@dataclass
class SSEEvent:
    id: int
    event: str
    data: dict
    created_at: float


@dataclass
class _Subscriber:
    queue: asyncio.Queue
    alive: bool = True
    last_heartbeat: float = field(default_factory=time.time)


class SSEBroadcaster:
    def __init__(self, history_size: int = 256, queue_max: int = 64) -> None:
        self._history: list[SSEEvent] = []
        self._history_size = history_size
        self._queue_max = queue_max
        self._subscribers: list[_Subscriber] = []
        self._next_id: int = 0
        self._lock = asyncio.Lock()
        self._hb_task: Optional[asyncio.Task] = None

    # ------------------------------------------------------------------
    # Lifecycle
    # ------------------------------------------------------------------

    async def start(self) -> None:
        if self._hb_task is None:
            self._hb_task = asyncio.create_task(self._heartbeat_loop())

    async def stop(self) -> None:
        if self._hb_task is not None:
            self._hb_task.cancel()
            try:
                await self._hb_task
            except (asyncio.CancelledError, Exception):
                pass
            self._hb_task = None
        for sub in list(self._subscribers):
            sub.alive = False
        self._subscribers.clear()

    # ------------------------------------------------------------------
    # Publish / Subscribe
    # ------------------------------------------------------------------

    def publish(self, event: str, data: dict) -> int:
        """Synchronous publish; returns assigned event id. Threadsafe in
        that asyncio cooperative scheduling serializes calls inside the
        loop — callers must NOT invoke from a thread."""
        self._next_id += 1
        ev = SSEEvent(
            id=self._next_id,
            event=event,
            data=data,
            created_at=time.time(),
        )
        self._history.append(ev)
        if len(self._history) > self._history_size:
            self._history = self._history[-self._history_size:]
        for sub in list(self._subscribers):
            if not sub.alive:
                self._subscribers.remove(sub)
                continue
            try:
                sub.queue.put_nowait(ev)
            except asyncio.QueueFull:
                # Slow subscriber — drop them so we don't block.
                sub.alive = False
                self._subscribers.remove(sub)
        return ev.id

    async def subscribe(
        self, last_event_id: Optional[int]
    ) -> AsyncIterator[SSEEvent]:
        """Async generator yielding events for this subscriber.

        1. If last_event_id is set, replay any history events newer than that
           id (in order) before yielding live events.
        2. Yield live events as they are published.
        3. Yield a periodic heartbeat event when no live activity occurs.
        """
        await self.start()
        sub = _Subscriber(queue=asyncio.Queue(maxsize=self._queue_max))
        async with self._lock:
            self._subscribers.append(sub)

        try:
            # 1. Replay history newer than last_event_id
            if last_event_id is not None:
                for ev in self._history:
                    if ev.id > last_event_id:
                        yield ev

            # 2. Live loop — rotate between queue.get and a short sleep that
            # checks for heartbeat timeout.
            while True:
                try:
                    ev = await asyncio.wait_for(sub.queue.get(), timeout=HEARTBEAT_INTERVAL)
                except asyncio.TimeoutError:
                    sub.last_heartbeat = time.time()
                    yield SSEEvent(id=0, event="heartbeat", data={}, created_at=time.time())
                    continue
                yield ev
        finally:
            sub.alive = False
            async with self._lock:
                if sub in self._subscribers:
                    self._subscribers.remove(sub)

    @property
    def history(self) -> list[SSEEvent]:
        return list(self._history)

    # ------------------------------------------------------------------
    # Heartbeat
    # ------------------------------------------------------------------

    async def _heartbeat_loop(self) -> None:
        while True:
            await asyncio.sleep(HEARTBEAT_INTERVAL)
            for sub in list(self._subscribers):
                if not sub.alive:
                    self._subscribers.remove(sub)
                    continue
                try:
                    sub.queue.put_nowait(
                        SSEEvent(id=0, event="heartbeat", data={}, created_at=time.time())
                    )
                except asyncio.QueueFull:
                    sub.alive = False
                    self._subscribers.remove(sub)
```

```python
    proj = dir / project_id

def _parse_sse(raw: bytes) -> list[dict]:
    """Tolerate either 'data: {...}\\n\\n' or multi-line data blocks.

    Returns list of {id?, event?, data?} dicts in source order.
    """
    out = []
    for block in raw.decode("utf-8").split("\n\n"):
        ev: dict = {"data_parts": []}
        if not block.strip():
            continue
        for line in block.splitlines():
            if line.startswith(":"):
                continue
            if ":" not in line:
                continue
            k, _, v = line.partition(":")
            v = v.lstrip()
            if k == "id":
                ev["id"] = int(v)
            elif k == "event":
                ev["event"] = v
            elif k == "data":
                ev["data_parts"].append(v)
        if "data_parts" in ev and ev["data_parts"]:
            ev["data"] = "\n".join(ev["data_parts"])
            del ev["data_parts"]
        if ev:
            out.append(ev)
    return out
```

```python
    async def session_events(
        project_id: str,
        request: Request,
        last_event_id: Optional[int] = Header(None, alias="Last-Event-ID"),
    ):
        snap = fm.read_json(project_id, str(SESSION_REL))
        if not snap:
            raise HTTPException(status_code=404, detail={"error": True, "code": "NO_SESSION"})

        async def event_stream() -> AsyncIterator[bytes]:
            # 1. Send snapshot immediately
            yield _format_sse("snapshot", snap, id_=None)
            # 2. Subscribe to live
            async for ev in broadcaster.subscribe(last_event_id):
                if await request.is_disconnected():
                    break
                if ev.event == "heartbeat":
                    yield b":hb\n\n"
                    continue
                yield _format_sse(ev.event, ev.data, id_=ev.id)
                # Cooperative yield so the response coroutine can be cancelled
                await asyncio.sleep(0)

        return StreamingResponse(
            event_stream(), media_type="text/event-stream"
        )

    return router


def _format_sse(event: str, data: dict, id_: Optional[int]) -> bytes:
    payload = json.dumps(data, ensure_ascii=False)
    chunks = []
    if id_ is not None:
        chunks.append(f"id: {id_}")
    chunks.append(f"event: {event}")
    chunks.append(f"data: {payload}")
    chunks.append("")  # blank line terminator
    chunks.append("")
    return ("\n".join(chunks) + "\n").encode("utf-8")
```

**Context:** The browser-native `EventSource` API does NOT support custom request headers — `Last-Event-ID` is sent automatically by the browser only when the server sends `id:` lines AND the connection drops, on the **reconnect** GET it sends `Last-Event-ID` automatically. So we only need manual reconnect for our **exponential backoff with stored id from a known location** (we capture id from the most recent received event).

This task creates a thin typed wrapper that:

1. Opens an `EventSource` to `/api/v1/projects/{id}/autopilot/session/events`.
2. Emits events as `{ id, event, data }` through a callback interface.
3. Tracks the latest event id in `localStorage` so a manual reconnect (after our backoff decision) can set `EventSource` headers... but wait — **EventSource does not allow headers**. The browser automatically sends `Last-Event-ID` on reconnect only if the server set `id:` fields. So our job here is purely:
   - Open connection.
   - Forward events to subscribers.
   - Surface connection state changes (open/error/close) so the hook can decide to manually re-open after backoff.

The reconnect-on-error flow lives in Task 4 (`useAutopilotSession`); the client is just the transport.

```ts
/** Typed wrapper around the browser-native EventSource for the
 * AutopilotSession event stream.
 *
 * NOTE: EventSource does not allow setting the Last-Event-ID header
 * manually — browsers send it automatically on reconnect when the server
 * set `id:` lines. We surface an `onError` so callers can decide whether to
 * manually re-open (the hook in useAutopilotSession.ts implements
 * exponential backoff and re-opens).
 */
export interface AutopilotEvent {
  event: string;
  data: unknown;
  id?: number;
}

export interface SSEHandle {
  close: () => void;
}

export interface SSEHandlers {
  onEvent: (ev: AutopilotEvent) => void;
  onOpen?: () => void;
  onError?: () => void;
}

export function connectAutopilotSSE(
  projectId: string,
  handlers: SSEHandlers,
): SSEHandle {
  const url = `/api/v1/projects/${encodeURIComponent(projectId)}/autopilot/session/events`;
  const source = new EventSource(url);

  source.addEventListener("open", () => handlers.onOpen?.());
  source.addEventListener("error", () => handlers.onError?.());

  // Use addEventListener instead of `onmessage` so we can route by event type.
  const typedEvents = [
    "snapshot",
    "session_start",
    "task_start",
    "task_complete",
    "task_fail",
    "decision",
    "circuit_open",
    "circuit_close",
    "queue_add",
    "queue_remove",
  ];

  for (const ev of typedEvents) {
    source.addEventListener(ev, (raw) => {
      const me = raw as MessageEvent;
      let data: unknown = me.data;
      try {
        data = JSON.parse(me.data);
      } catch {
        // not JSON — leave as string
      }
      const id = me.lastEventId ? Number(me.lastEventId) : undefined;
      handlers.onEvent({ event: ev, data, id: Number.isFinite(id) ? id : undefined });
    });
  }

  return {
    close() {
      source.close();
    },
  };
}
```

```ts
import { useEffect, useRef, useState, useCallback } from "react";
import {
  AutopilotEvent,
  AutopilotSession,
  ManagedStartConfig,
  connectAutopilotSSE,
  getAutopilotSession,
  startAutopilotSession,
  stopAutopilotSession,
  pauseAutopilotSession,
  resumeAutopilotSession,
} from "../api/autopilot";

export type SSEStatus = "connecting" | "connected" | "reconnecting" | "error";

interface UseAutopilotSessionReturn {
  session: AutopilotSession | null;
  events: AutopilotEvent[];
  status: SSEStatus;
  start: (config: ManagedStartConfig) => Promise<void>;
  stop: () => Promise<void>;
  pause: () => Promise<void>;
  resume: () => Promise<void>;
  refresh: () => Promise<void>;
}

const BACKOFF_DELAYS_MS = [1000, 2000, 4000, 8000, 16000, 30000];

export function useAutopilotSession(
  projectId: string,
): UseAutopilotSessionReturn {
  const [session, setSession] = useState<AutopilotSession | null>(null);
  const [events, setEvents] = useState<AutopilotEvent[]>([]);
  const [status, setStatus] = useState<SSEStatus>("connecting");
  const reconnectAttempts = useRef(0);
  const handleRef = useRef<{ close: () => void } | null>(null);
  const cancelledRef = useRef(false);

  const refresh = useCallback(async () => {
    const s = await getAutopilotSession(projectId);
    if (!cancelledRef.current) setSession(s);
  }, [projectId]);

  const open = useCallback(() => {
    if (cancelledRef.current) return;
    setStatus(reconnectAttempts.current > 0 ? "reconnecting" : "connecting");
    const handle = connectAutopilotSSE(projectId, {
      onEvent: (ev) => {
        // Snapshot events replace session state; non-snapshot events append
        setEvents((prev) => [...prev, ev].slice(-256));
        if (ev.event === "snapshot" && ev.data && typeof ev.data === "object") {
          setSession(ev.data as AutopilotSession);
        }
        setStatus("connected");
        reconnectAttempts.current = 0;
      },
      onOpen: () => setStatus("connected"),
      onError: () => {
        handle.close();
        const delay =
          BACKOFF_DELAYS_MS[
            Math.min(reconnectAttempts.current, BACKOFF_DELAYS_MS.length - 1)
          ];
        reconnectAttempts.current += 1;
        setStatus("reconnecting");
        setTimeout(() => {
          if (!cancelledRef.current) open();
        }, delay);
      },
    });
    handleRef.current = handle;
  }, [projectId]);

  useEffect(() => {
    cancelledRef.current = false;
    refresh();
    open();
    return () => {
      cancelledRef.current = true;
      handleRef.current?.close();
    };
  }, [refresh, open]);

  const start = useCallback(
    async (config: ManagedStartConfig) => {
      const next = await startAutopilotSession(projectId, config);
      setSession(next);
    },
    [projectId],
  );

  const stop = useCallback(async () => {
    await stopAutopilotSession(projectId);
    await refresh();
  }, [projectId, refresh]);

  const pause = useCallback(async () => {
    await pauseAutopilotSession(projectId);
    await refresh();
  }, [projectId, refresh]);

  const resume = useCallback(async () => {
    await resumeAutopilotSession(projectId);
    await refresh();
  }, [projectId, refresh]);

  return { session, events, status, start, stop, pause, resume, refresh };
}
```

```ts
import type { ManagedStartConfig } from "../components/workspace/ManagedStartModal";
import {
  getAutopilotSession,
  startAutopilotSession,
} from "../api/autopilot";

const FALLBACK: ManagedStartConfig = {
  scope: "all_planned",
  cadence: "balanced",
  policy: "auto",
  notify: "milestones",
};

export function useAutopilotConfig(projectId: string) {
  const [config, setConfig] = useState<ManagedStartConfig>(FALLBACK);
  const [loaded, setLoaded] = useState(false);
  const [submitting, setSubmitting] = useState(false);

  useEffect(() => {
    let cancelled = false;
    getAutopilotSession(projectId)
      .then((s) => {
        if (cancelled) return;
        if (s?.config) setConfig(s.config);
        setLoaded(true);
      })
      .catch(() => setLoaded(true));
    return () => {
      cancelled = true;
    };
  }, [projectId]);

  const submit = useCallback(async () => {
    setSubmitting(true);
    try {
      await startAutopilotSession(projectId, config);
    } finally {
      setSubmitting(false);
    }
  }, [projectId, config]);

  return { config, setConfig, loaded, submitting, submit, defaults: FALLBACK };
}
```

```python
from fastapi import APIRouter
import json
from pathlib import Path
from backend.config import settings

router = APIRouter(prefix="/api/v1/users", tags=["users"])
PREFS = Path(settings.projects_dir).parent / "user_preferences.json"


def _read() -> dict:
    if not PREFS.exists():
        return {"workspace_mode": None, "workspace_panel": None}
    return json.loads(PREFS.read_text(encoding="utf-8"))


def _write(payload: dict) -> None:
    PREFS.parent.mkdir(parents=True, exist_ok=True)
    PREFS.write_text(json.dumps(payload, ensure_ascii=False), encoding="utf-8")


@router.get("/me/preferences")
async def get_preferences():
    return {"error": False, "code": "OK", "message": "", "detail": _read()}


@router.put("/me/preferences")
async def put_preferences(body: dict):
    payload = _read()
    if "workspace_mode" in body:
        payload["workspace_mode"] = body["workspace_mode"]
    if "workspace_panel" in body:
        payload["workspace_panel"] = body["workspace_panel"]
    _write(payload)
    return {"error": False, "code": "OK", "message": "", "detail": payload}
```
